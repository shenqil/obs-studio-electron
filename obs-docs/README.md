## Quick Start Example

```javascript
// Basic OBS setup pattern
obs.OBS_API_initAPI();

try {
  // Create scene
  const scene = obs.osn.Scene.Create("Main Scene");
  
  // Add source
  const source = obs.osn.Input.Create("video_capture", "Webcam", {});
  scene.AddSource(source.CallGetId());
  
  // Configure streaming
  const service = obs.osn.Service.Create("rtmp_common", "Twitch", settings);
  const streaming = new obs.osn.Streaming();
  streaming.SetService(service);
  streaming.Start();
  
  // Clean up
  source.CallRelease();
  scene.CallRelease();
  
} finally {
  obs.OBS_API_destroyOBS_API();
}
```
